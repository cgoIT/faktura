  COLUMN document_id character varying(36)
  COLUMN id character varying(36) NOT NULL
  COLUMN plan_id character varying(36) NOT NULL
  COLUMN service_from character varying(10) NOT NULL
  COLUMN service_to character varying(10) NOT NULL
  COLUMN status character varying(10) NOT NULL
  FOREIGN KEY (plan_id) REFERENCES recurring_plan(id)
  PRIMARY KEY (id)
  UNIQUE (plan_id, service_from)
  ck_recurring_run_status CHECK (((status)::text = ANY ((ARRAY['DRAFT'::character varying, 'RELEASED'::character varying, 'SKIPPED'::character varying])::text[])))
  INDEX CREATE INDEX ix_recurring_run_document ON public.recurring_run USING btree (document_id)
  INDEX CREATE INDEX ix_recurring_run_status ON public.recurring_run USING btree (status)

TABLE setup_step
  COLUMN done_at character varying(30) NOT NULL
  COLUMN step character varying(30) NOT NULL
  PRIMARY KEY (step)

TABLE shipping_option
  COLUMN active integer NOT NULL DEFAULT 1
  COLUMN amount_cents bigint NOT NULL DEFAULT 0
  COLUMN created_at character varying(30) NOT NULL
  COLUMN id character varying(36) NOT NULL
  COLUMN name character varying(100) NOT NULL
  COLUMN sort_order integer NOT NULL DEFAULT 0
  COLUMN tax_category character varying(2)
  COLUMN tax_rate_bp integer NOT NULL DEFAULT 0
  PRIMARY KEY (id)
  ck_shipping_option_active CHECK ((active = ANY (ARRAY[0, 1])))
  ck_shipping_option_amount CHECK ((amount_cents >= 0))
  ck_shipping_option_rate CHECK (((tax_rate_bp >= 0) AND (tax_rate_bp <= 10000)))

TABLE tax_category_reason
  COLUMN category character varying(2) NOT NULL
  COLUMN reason character varying(500) NOT NULL
  PRIMARY KEY (category)

TABLE tax_profile
  COLUMN allowed_categories character varying(100)
  COLUMN company_id character varying(36) NOT NULL
  COLUMN default_category character varying(2) NOT NULL DEFAULT 'E'::character varying
  COLUMN default_rate_bp integer NOT NULL DEFAULT 0
  COLUMN exemption_text character varying(500)
  COLUMN id character varying(36) NOT NULL
  COLUMN notice_text character varying(500)
  COLUMN regime character varying(20) NOT NULL
  COLUMN valid_from character varying(10) NOT NULL
  COLUMN valid_to character varying(10)
  FOREIGN KEY (company_id) REFERENCES company(id)
  PRIMARY KEY (id)
